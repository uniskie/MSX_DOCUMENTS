    ; Nextor 2.1 用のダミーディスクドライバー
    ; By Konamiman, 8/2018
    ;
    ; このコードは、実際のディスクドライバーを開発する際のベースとして
    ; 使用できます。DRV_TYPE を適切に設定し、DRV_NAME や VER_* の
    ; ドライバー名とバージョンを変更し、ドライバータイプに応じた
    ; 必要なルーチンを実装してください。
    ;
    ; 詳細については、Nextor ドライバー開発ガイドを参照してください。

```
    ;org 4100h      DRV_START:  

;-----------------------------------------------------------------------------
;
; 各種定数
;

;これは実行するコードのアドレスを格納するための2バイトのバッファです。
;一部のカーネルページ0ルーチンで使用されます。

CODE_ADD:	equ	0F1D0h


;-----------------------------------------------------------------------------
;
; ドライバー設定定数
;

;ドライバータイプ:
;   0: ドライブベース
;   1: デバイスベース

DRV_TYPE	equ	0


;ドライバーバージョン

VER_MAIN	equ	1
VER_SEC		equ	0
VER_REV		equ	0


;-----------------------------------------------------------------------------
;
; DEV_RW 用のエラーコード
;

if DRV_TYPE eq 1

.NCOMP	equ	0FFh
.WRERR	equ	0FEh
.DISK	equ	0FDh
.NRDY	equ	0FCh
.DATA	equ	0FAh
.RNF	equ	0F9h
.WPROT	equ	0F8h
.UFORM	equ	0F7h
.SEEK	equ	0F3h
.IFORM	equ	0F0h
.IDEVL	equ	0B5h
.IPARM	equ	08Bh

endif


;-----------------------------------------------------------------------------
;
; カーネルページ0で利用可能なルーチンおよび情報
;

;* ページ1の現在のスロットをAに取得します。Fレジスタは破壊されます。
;  バンク0に対してCALBNKを使用して呼び出す必要があります:
;    xor a
;    ld ix,GSLOT1
;    call CALBNK

GSLOT1	equ	402Dh


;* このルーチンは、別のバンクから1バイトを読み取ります。
;  対象のバンクに対してCALBNKを使用し、読み取るアドレスを
;  HLに渡して呼び出す必要があります:
;    ld a,<バンク番号>
;    ld hl,<バイトアドレス>
;    ld ix,RDBANK
;    call CALBNK

RDBANK	equ	403Ch


;* このルーチンは、一時的にカーネルのメインバンク
;  （通常はバンク0ですが、MSX-DOS 1モードで動作している場合は3になります）
;  を切り替え、(CODE_ADD)にあるアドレスのルーチンを呼び出します。
;  CALBASを呼び出す際（BASICエラー発生時にカーネルバンクが正しくなるようにするため）
;  や、F37Dhフック経由でDOS機能を呼び出す際には、このルーチンを使用する
;  必要があります。
;
;  入力:  (CODE_ADD) に呼び出すコードのアドレス。
;          AF, BC, DE, HL, IX, IY は呼び出されるルーチンにそのまま渡されます。
;  出力:  AF, BC, DE, HL, IX, IY は呼び出されたルーチンから返されます。

CALLB0	equ	403Fh


;* 別のバンクのルーチンを呼び出します。
;  ドライバーが複数のバンクにまたがる場合に使用する必要があります。
;
;  入力:  A = バンク番号
;          IX = ルーチンのアドレス
;          AF' = ルーチン用の AF
;          HL' = ルーチン用の IX
;          BC, DE, HL, IY = ルーチンへの入力
;  出力:  AF, BC, DE, HL, IX, IY は呼び出されたルーチンから返されます。

CALBNK	equ	4042h


;* Aに渡されたスロットのSLTWRKエントリーのアドレスをIXに取得します。
;  このエントリーには、そのスロット用に割り当てられたページ3のワークエリア
;  へのポインタが格納されています（割り当てられていない場合は0）。
;  A=0 の場合、現在ページ1に切り替えられているスロットを使用します。
;  A=0 が渡された場合、Aにページ1の現在のスロットを返します。
;  Fレジスタは破壊されます。
;  バンク0に対してCALBNKを使用して呼び出す必要があります:
;    ld a,<スロット番号> (現在のページ1スロットの場合は xor a)
;    ex af,af'
;    xor a
;    ld ix,GWORK
;    call CALBNK

GWORK	equ	4045h


;* このアドレスには、Nextorカーネルを構成するバンク数
;  （または、ドライバーの開始バンク番号）を示す1バイトが含まれています。

K_SIZE	equ	40FEh


;* このアドレスには、現在のバンク番号を示す1バイトが含まれています。

CUR_BANK	equ	40FFh


;-----------------------------------------------------------------------------
;
; 内蔵フォーマット選択文字列
;

NULL_MSG  equ     781Fh	;ヌル文字列（ディスクのフォーマット不可）
SING_DBL  equ     7820h ;"1-Single side / 2-Double side"


;-----------------------------------------------------------------------------
;
; ドライバー署名
;
	db	"NEXTOR_DRIVER",0


;-----------------------------------------------------------------------------
;
; ドライバーフラグ:
;    bit 0: 0はドライブベース、1はデバイスベース
;    bit 2: ドライバーが DRV_CONFIG ルーチンを実装している場合は 1
;             (Nextor v2.0.5 以降で使用)

if DRV_TYPE eq 0
	db	0+4
endif

if DRV_TYPE eq 1
	db	1+4
endif


;-----------------------------------------------------------------------------
;
; 予約バイト
;

	db	0


;-----------------------------------------------------------------------------
;
; ドライバー名
;

DRV_NAME:
	db	"Dummy driver"
	ds	32-($-DRV_NAME)," "


;-----------------------------------------------------------------------------
;
; ドライバーの公開ルーチンのジャンプテーブル
;

	; これらのルーチンはすべてのドライバーで必須です
        ; (ただし、通常実装する必要があるのは DRV_INIT のみです)

	jp	DRV_TIMI
	jp	DRV_VERSION
	jp	DRV_INIT
	jp	DRV_BASSTAT
	jp	DRV_BASDEV
    jp  DRV_EXTBIO
    jp  DRV_DIRECT0
    jp  DRV_DIRECT1
    jp  DRV_DIRECT2
    jp  DRV_DIRECT3
    jp  DRV_DIRECT4
	jp	DRV_CONFIG

	ds	12

if DRV_TYPE eq 0

; これらのルーチンはドライブベースのドライバーで必須です

    jp  DRV_DSKIO
    jp  DRV_DSKCHG
    jp  DRV_GETDPB
    jp  DRV_CHOICE
    jp  DRV_DSKFMT
    jp  DRV_MTOFF
endif

if DRV_TYPE eq 1

	; これらのルーチンはデバイスベースのドライバーで必須です

	jp	DEV_RW
	jp	DEV_INFO
	jp	DEV_STATUS
	jp	LUN_INFO
endif


;=====
;=====  固定アドレスに配置しなければならないデータの終端
;=====


;-----------------------------------------------------------------------------
;
; タイマー割り込みルーチン。各タイマー割り込み（50Hzまたは60Hz）で
; 呼び出されますが、DRV_INIT の最初の実行時に Cy=1 が返された場合のみ呼び出されます。

DRV_TIMI:
	ret


;-----------------------------------------------------------------------------
;
; ドライバー初期化ルーチン。2回呼び出されます:
;
; 1) 1回目の実行（情報収集用）
;    入力:
;      A = 0
;      B = 使用可能なドライブ数
;      HL = ページ3で割り当て可能なワークエリアの最大サイズ
;      C: ユーザーがドライブ数の削減を要求している場合はビット5がセットされる
;         (5キーを押しながら起動した場合)
;    出力:
;      A = 必要なドライブ数（ドライブベースのドライバーのみ）
;      HL = ページ3で必要なワークエリアのサイズ
;      Cy = DRV_TIMI をタイマー割り込みにフックする場合は 1、そうでない場合は 0
;
; 2) 2回目の実行（ワークエリアおよびハードウェアの初期化用）
;    入力:
;      A = 1
;      B = このコントローラに割り当てられたドライブ数
;      C: ユーザーがドライブ数の削減を要求している場合はビット5がセットされる
;         (5キーを押しながら起動した場合)
;
;    ワークエリアのアドレスは GWORK を使用して取得できます。
;
;    1回目の実行で、使用可能な量を超えるワークエリアを要求した場合、
;    2回目の実行は行われず、DRV_TIMI もタイマー割り込みにフックされません。
;
;    1回目の実行で、使用可能な数を超えるドライブを要求した場合、
;    可能な限りのドライブが割り当てられ、初期化処理は通常通り続行されます。
;    (ドライブベースのドライバーのみ。デバイスベースのドライバーは常に
;     2つのドライブが割り当てられます。)

DRV_INIT:
	xor	a
	ld	hl,0
	ret


;-----------------------------------------------------------------------------
;
; ドライバーバージョンの取得
;
; 入力:  -
; 出力:  A = メジャーバージョン番号
;         B = マイナーバージョン番号
;         C = リビジョン番号

DRV_VERSION:
	ld	a,VER_MAIN
	ld	b,VER_SEC
	ld	c,VER_REV
	ret


;-----------------------------------------------------------------------------
;
; BASIC拡張ステートメント（"CALL"）ハンドラ。
; 通常通りに動作しますが、CALBASを呼び出す必要がある場合は、
; カーネルページ0の CALLB0 ルーチンを経由して行う必要があります。

DRV_BASSTAT:
	scf
	ret


;-----------------------------------------------------------------------------
;
; BASIC拡張デバイスハンドラ。
; 通常通りに動作しますが、CALBASを呼び出す必要がある場合は、
; カーネルページ0の CALLB0 ルーチンを経由して行う必要があります。

DRV_BASDEV:
	scf
	ret


;-----------------------------------------------------------------------------
;
; 拡張BIOSフック。
; 通常通りに動作しますが、古いフックを呼び出す必要がある場合は D'=1、
; そうでない場合は D'=0 を返す必要があります。
; 呼び出し時は D'=1 で入ります。

DRV_EXTBIO:
	ret


;-----------------------------------------------------------------------------
;
; ダイレクトコールエントリーポイント。
; カーネルバンク0および3のアドレス 7850h, 7853h, 7856h, 7859h, 785Ch
; への呼び出しは、それぞれ DIRECT0/1/2/3/4 にリダイレクトされます。
; IX および AF' を除くすべてのレジスタデータを呼び出し元から受け取ります。

DRV_DIRECT0:
DRV_DIRECT1:
DRV_DIRECT2:
DRV_DIRECT3:
DRV_DIRECT4:
	ret


;-----------------------------------------------------------------------------
;
; ドライバー設定の取得
; (このルーチンを実装する場合、ドライバーフラグのビット2をセットする必要があります)
;
; 入力:
;   A = 設定インデックス
;   BC, DE, HL = 設定内容に依存
;
; 出力:
;   A = 0: 成功
;       1: 指定されたインデックスの設定は利用不可
;   BC, DE, HL = 設定内容に依存
;
; * 起動時のドライブ数の取得（デバイスベースのドライバーのみ）:
;   入力:
;     A = 1
;     B = DOS 2 モードの場合は 0、DOS 1 モードの場合は 1
;     C: ユーザーがドライブ数の削減を要求している場合はビット5がセットされる
;        (5キーを押しながら起動した場合)
;   出力:
;     B = ドライブ数
;
; * ドライブのデフォルト設定の取得
;   入力:
;     A = 2
;     B = DOS 2 モードの場合は 0、DOS 1 モードの場合は 1
;     C = 起動時の相対ドライブ番号
;   出力:
;     B = デバイスインデックス
;     C = LUNインデックス

DRV_CONFIG:
	ld a,1
	ret
	

;=====
;=====  ドライブベース専用ルーチンの開始
;=====

if DRV_TYPE eq 0

;-----------------------------------------------------------------------------
;
; ディスクセクタの読み込み/書き込み
;
;入力:    A  = ドライブ番号（0から開始）
;          Cy = 0: セクタの読み込み、1: セクタの書き込み
;          B  = 読み込み/書き込みを行うセクタ数
;          C  = ビット7 = 0 の場合、読み込み/書き込みを行う最初のセクタ番号（ビット22-16）
;               ビット7 = 1 の場合、メディアID
;          DE = 読み込み/書き込みを行う最初のセクタ番号（ビット15-0）
;          HL = 転送元の転送先/転送元アドレス
;出力:    Cy = 0: 成功、1: エラー
;          A  = エラーコード（エラー時のみ）:
;               0   書き込み禁止（ライトプロテクト）
;               2   準備未完了（ノットレディ）
;               4   データ (CRC) エラー
;               6   シークエラー
;               8   レコード未検出（レコードノットファウンド）
;               10  書き込み不具合（ライトフォルト）
;               12  その他のエラー
;          B = 実際に読み込まれたセクタ数（エラー時のみ）

DRV_DSKIO:
	ld	a,12
	scf
	ret


;-----------------------------------------------------------------------------
;
; ディスク変更状態の取得
;
;入力:    A  = ドライブ番号（0から開始）
;          B  = C = メディア記述子
;          HL = DPB -1 のベースアドレス
;出力:    Cy = 0: 成功、1: エラー
;          A  = エラーコード（エラー時のみ）
;               DRV_DSKIO と同じコード
;          B  = ディスク状態（成功時のみ）
;               1  ディスク変更なし
;               0  不明
;              -1  ディスク変更あり

DRV_DSKCHG:
	ld	a,12
	scf
      	ret
      
      
;-----------------------------------------------------------------------------
;
; ディスクの DPB の取得
;
;入力:    A  = ドライブ番号（0から開始）
;          B  = C = メディア記述子
;          HL = DPB -1 のベースアドレス
;出力:    -

DRV_GETDPB:
	ld	a,12
	scf
	ret


;-----------------------------------------------------------------------------
;
; フォーマット選択文字列の返却
;
;入力:   -
;出力:  HL = カーネルスロット内の選択文字列のアドレス

DRV_CHOICE:
	ld      hl,NULL_MSG
	ret


;-----------------------------------------------------------------------------
;
; ディスクのフォーマット
;
;入力:   A  = フォーマットの選択肢、1から9（DRV_CHOICE を参照）
;         D  = ドライブ番号（0から開始）
;         HL = メモリ内のワークエリアのアドレス
;         DE = ワークエリアのサイズ
;出力:  Cy = 0: 成功、1: エラー
;         A  = エラーコード（エラー時のみ）:
;              0   書き込み禁止（ライトプロテクト）
;              2   準備未完了（ノットレディ）
;              4   データ (CRC) エラー
;              6   シークエラー
;              8   レコード未検出（レコードノットファウンド）
;              10  書き込み不具合（ライトフォルト）
;              12  不正なパラメータ
;              14  メモリ不足
;              16  その他のエラー

DRV_DSKFMT:
	ld	a,16
	scf
	ret


;-----------------------------------------------------------------------------
;
; フロッピーディスクドライブのモーターをオフにする
;
;入力:   -
;出力:  -

DRV_MTOFF:
	ret

endif


;=====
;=====  ドライブベース専用ルーチンの終了
;=====


;=====
;=====  デバイスベース専用ルーチンの開始
;=====

if DRV_TYPE eq 1

;-----------------------------------------------------------------------------
;
; 論理ユニットに対する論理セクタの読み込みまたは書き込み
;
;入力:    Cy=0: 読み込み、1: 書き込み
;          A = デバイス番号（1から7）
;          B = 読み込みまたは書き込みを行うセクタ数
;          C = 論理ユニット番号（1から7）
;          HL = 転送元の転送先または転送元メモリ（RAM）アドレス
;          DE = 4バイトのセクタ番号が格納されているアドレス
;出力:    A = エラーコード（MSX-DOSと同じコードが使用されます）:
;              0: 成功
;              .IDEVL: 無効なデバイスまたはLUN
;              .NRDY: 準備未完了（ノットレディ）
;              .DISK: 一般的な不明なディスクエラー
;              .DATA: 読み込み時のCRCエラー
;              .RNF: セクタ未検出（セクタノットファウンド）
;              .UFORM: 未フォーマットのディスク
;              .WPROT: 書き込み禁止メディア、または読み取り専用論理ユニット
;              .WRERR: 書き込みエラー
;              .NCOMP: 非互換ディスク
;              .SEEK: シークエラー
          B = 実際に読み込まれたセクタ数（エラー時のみ）

DEV_RW:
	ld	a,.NRDY
	ld	b,0
	ret


;-----------------------------------------------------------------------------
;
; デバイス情報の収集
;
;入力:   A = デバイスインデックス（1から7）
;         B = 返す情報の内容:
;             0: 基本情報
;             1: 製造元名文字列
;             2: デバイス名文字列
;             3: シリアル番号文字列
;         HL = RAM内のバッファへのポインタ
;出力:  A = エラーコード:
;             0: 成功
;             1: デバイスが利用不可、または無効なデバイスインデックス
;             2: 情報が利用不可、または無効な情報インデックス
;         基本情報が要求された場合、
;         バッファは以下の情報で満たされます:
;
;+0 (1): 論理ユニットの数（1から7）。デバイスに論理ユニットがない場合は 1
;        （これは実質的に1つの論理ユニットがある場合と等価です）。
;+1 (1): デバイスフラグ（Beta 2では常に0）。
;
; 文字列は印刷可能なASCII文字列（ASCIIコード32から126）で、
; 左詰めでスペースでパディングされている必要があります。すべての文字列は任意であり、
; 利用できない場合はエラーを返す必要があります。
; デバイスからバイナリ形式で文字列が提供される場合は、
; プレフィックス "0x" を付加した大文字の16進数文字列として報告する必要があります。
; 文字列の最大長は64文字です。
; 文字列が実際にはこれより長い場合は、左端の64文字を
; 提供する必要があります。
;
; シリアル番号文字列の場合も同じルールが適用されますが、
; 右詰めで提供される必要があり、長すぎる場合は左端ではなく
; 右端の文字群を提供する必要があります。

DEV_INFO:
	ld	a,1
	ret


;-----------------------------------------------------------------------------
;
; デバイス状態の取得
;
;入力:   A = デバイスインデックス（1から7）
;         B = 論理ユニット番号（1から7）
;             0 を指定するとデバイス自体の状態を返します。
;出力:  A = 指定された論理ユニットの状態、
;             または0が指定された場合はデバイス全体の状態:
;                0: デバイスまたは論理ユニットが利用不可、あるいは
;                   指定されたデバイス番号または論理ユニット番号が無効。
;                1: デバイスまたは論理ユニットが利用可能であり、
;                   前回の状態要求から変更されていない。
;                2: デバイスまたは論理ユニットが利用可能であり、
;                   前回の状態要求から変更されている
;                   （デバイスの場合、デバイスが抜かれ、同じデバイス
;                    インデックスが割り当てられた別のデバイスが挿入されたこと、
;                    論理ユニットの場合、メディアが変更されたことを意味します）。
;                3: デバイスまたは論理ユニットは利用可能だが、前回の
;                   状態要求から変更されたかどうかを判断できない。
;
; ホットプラグをサポートしていないデバイスは、常に状態値として 1 を返す必要があります。
; 取り外し不可能な論理ユニットは 0 および 1 を返す場合があります。
;
; 返される状態は常に DEV_STATUS自体の「前回の呼び出し」に対して相対的です。
; 詳細については、ドライバー開発者ガイドを参照してください。

DEV_STATUS:
	xor	a
	ret


;-----------------------------------------------------------------------------
;
; 論理ユニット情報の取得
;
;入力:   A  = デバイスインデックス（1から7）
;         B  = 論理ユニット番号（1から7）
;         HL = RAM内のバッファへのポインタ
;出力:  A = 0: 成功、バッファに情報が格納されます。
;             1: エラー、デバイスまたは論理ユニットが利用不可、
;                または無効なデバイスインデックスもしくは論理ユニット番号。
;         成功時、バッファは以下の情報で満たされます:
;
;+0 (1): メディアタイプ:
;        0: ブロックデバイス
;        1: CD または DVD のリーダー/ライター
;        2-254: 未使用。将来的に追加のコードが定義される可能性があります。
;        255: その他
;+1 (2): セクタサイズ。この情報が適用されない、または
;        利用できない場合は 0。
;+3 (4): 利用可能な総セクタ数。
;        この情報が適用されない、または利用できない場合は 0。
;+7 (1): フラグ:
;        bit 0: メディアが取り外し可能（リムーバブル）な場合は 1。
;        bit 1: メディアが読み取り専用（リードオンリー）の場合は 1。動的に
;               書き込み禁止/解除を切り替えられるメディアは
;               読み取り専用とはみなされません。
;        bit 2: LUN がフロッピーディスクドライブである場合は 1。
;        bit 3: この LUN を自動マッピング（オートマッピング）に使用すべきでない場合は 1。
;        bit 4-7: 常に 0 でなければなりません。
;+8 (2): シリンダ数
;+10 (1): ヘッド数
;+11 (1): トラックあたりのセクタ数
;
; シリンダ数、ヘッド数、セクタ数はハードディスクにのみ適用されます。
; その他のタイプのデバイスでは、これらのフィールドは 0 にする必要があります。

LUN_INFO:
	ld	a,1
	ret

endif
```
